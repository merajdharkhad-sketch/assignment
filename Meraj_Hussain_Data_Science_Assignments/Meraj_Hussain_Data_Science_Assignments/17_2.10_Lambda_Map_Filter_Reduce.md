## Task 1: Use lambda with map to double values
```python
distances_km = [2, 5, 7, 10]
doubled_distances = list(map(lambda km: km * 2, distances_km))
print(doubled_distances)
```

## Task 2: Clean product names using lambda
```python
product_names = ["  cotton shirt ", "wireless MOUSE  ", "  steel bottle"]
clean_names = list(map(lambda title: title.strip().title(), product_names))
print(clean_names)
```

## Task 3: Filter names that match a search
```python
friend_names = ["Aarav Patel", "Nisha Shah", "Arjun Mehta", "Nisha Khan"]
search_word = "nisha"
matching_names = list(filter(lambda person: search_word in person.lower(), friend_names))
print(matching_names)
```

## Task 4: Find the average of ratings that are at least 4
```python
from functools import reduce

ratings = [3.5, 4.2, 4.8, 2.9, 4.0, 5.0]
good_ratings = list(filter(lambda score: score >= 4, ratings))
if good_ratings:
    rating_sum = reduce(lambda total, score: total + score, good_ratings)
    print("Average rating:", rating_sum / len(good_ratings))
else:
    print("No ratings of 4 or above")
```

## Task 5: Filter ratings and calculate their average using reduce
```python
from functools import reduce

product_ratings = [5, 3, 4, 2, 4, 5, 3]
high_ratings = list(filter(lambda score: score >= 4, product_ratings))
if high_ratings:
    sum_ratings = reduce(lambda total, score: total + score, high_ratings, 0)
    average_rating = sum_ratings / len(high_ratings)
    print("Ratings used:", high_ratings)
    print("Average:", average_rating)
else:
    print("No ratings passed the filter")
```
