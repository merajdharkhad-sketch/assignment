## Task 1: Mean of daily step counts
```python
def mean(arr):
    if not arr:
        return 0
    return sum(arr) / len(arr)

step_counts = [4200, 6100, 5300, 7800, 6600]
print("Mean steps:", mean(step_counts))
```

## Task 2: Median of order values
```python
def median(arr):
    if not arr:
        return None
    sorted_values = sorted(arr)
    middle = len(sorted_values) // 2
    if len(sorted_values) % 2 == 1:
        return sorted_values[middle]
    return (sorted_values[middle - 1] + sorted_values[middle]) / 2

order_values = [250, 400, 150, 600, 300]
print("Median order:", median(order_values))
```

## Task 3: Most frequent song duration
```python
from collections import Counter

def mode(arr):
    if not arr:
        return None
    counts = Counter(arr)
    highest_count = max(counts.values())
    return [value for value, count in counts.items() if count == highest_count]

song_seconds = [182, 210, 182, 195, 210, 182, 205]
print("Most common duration(s):", mode(song_seconds))
```

## Task 4: Compare three post-like lists
```python
from collections import Counter

def mean(arr):
    return sum(arr) / len(arr) if arr else 0

def median(arr):
    if not arr:
        return None
    values = sorted(arr)
    mid = len(values) // 2
    return values[mid] if len(values) % 2 else (values[mid - 1] + values[mid]) / 2

def mode(arr):
    if not arr:
        return None
    counts = Counter(arr)
    max_count = max(counts.values())
    return [value for value, count in counts.items() if count == max_count]

post_a = [100, 120, 110, 130, 115]
post_b = [80, 90, 95, 100, 500]
post_c = [60, 60, 70, 70, 80]
for label, likes in [("A", post_a), ("B", post_b), ("C", post_c)]:
    print(label, "mean:", mean(likes), "median:", median(likes), "mode:", mode(likes))
```
For list A, the mean and median both describe the typical post fairly well. For list B, the median is more representative because one unusually high value (500) pulls the mean upward. For list C, the mode shows the repeated like counts, while the median also gives a central value.
