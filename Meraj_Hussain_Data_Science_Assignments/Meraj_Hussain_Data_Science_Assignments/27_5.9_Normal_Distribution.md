## Task 1: Generate 4,000 normal random values
```python
import random
random.seed(12)
normal_values = [random.gauss(0, 1) for _ in range(4000)]
print("First 10 values:", normal_values[:10])
```

## Task 2: Plot 1,000 values and mark the mean
```python
import random
import matplotlib.pyplot as plt

sample = [random.gauss(0, 1) for _ in range(1000)]
sample_mean = sum(sample) / len(sample)
plt.hist(sample, bins=35, density=True, alpha=0.65)
plt.axvline(sample_mean, linestyle="--", label=f"Mean = {sample_mean:.2f}")
plt.title("Normal Distribution Sample")
plt.xlabel("Value")
plt.ylabel("Density")
plt.legend()
plt.show()
```

## Task 3: Percentage within 1, 2 and 3 standard deviations
```python
import math

avg = sum(normal_values) / len(normal_values)
std_dev = math.sqrt(sum((v - avg) ** 2 for v in normal_values) / len(normal_values))
for width in (1, 2, 3):
    lower, upper = avg - width * std_dev, avg + width * std_dev
    count = sum(lower <= value <= upper for value in normal_values)
    print(f"Within {width} standard deviation(s): {count / len(normal_values) * 100:.2f}%")
```
For a normal distribution, the approximate theoretical percentages are 68%, 95%, and 99.7%.

## Task 4: Shade the 68%, 95%, and 99.7% ranges
```python
import numpy as np
import matplotlib.pyplot as plt

x_axis = np.linspace(-4, 4, 600)
density = (1 / np.sqrt(2 * np.pi)) * np.exp(-(x_axis ** 2) / 2)
plt.plot(x_axis, density, label="Standard normal curve")
plt.fill_between(x_axis, density, where=(x_axis >= -1) & (x_axis <= 1), alpha=0.35, label="Within 1 SD")
plt.fill_between(x_axis, density, where=(x_axis >= -2) & (x_axis <= 2), alpha=0.18, label="Within 2 SD")
plt.fill_between(x_axis, density, where=(x_axis >= -3) & (x_axis <= 3), alpha=0.08, label="Within 3 SD")
plt.axvline(0, linestyle="--", label="Mean")
plt.legend()
plt.show()
```
